    <script>
        // 小学生低学年でもわかる・ひらがな表記のお題ペアリスト
        const WORD_PAIRS = [
            { citizen: "りんご", wolf: "みかん" },
            { citizen: "いぬ", wolf: "ねこ" },
            { citizen: "かれー", wolf: "しちゅー" },
            { citizen: "ぶらんこ", wolf: "すべりだい" },
            { citizen: "さっかー", wolf: "やきゅう" },
            { citizen: "えんぴつ", wolf: "けしごむ" },
            { citizen: "でんしゃ", wolf: "ひこうき" },
            { citizen: "どうぶつえん", wolf: "すいぞくかん" },
            { citizen: "らーめん", wolf: "うどん" },
            { citizen: "ぽてとちっぷす", wolf: "ぽっぷこーん" },
            { citizen: "えいがかん", wolf: "ゆうえんち" },
            { citizen: "すいか", wolf: "めろん" },
            { citizen: "ぱん", wolf: "ぐらたん" },
            { citizen: "ちよこれーと", wolf: "あめだま" },
            { citizen: "うみの いえ", wolf: "プール" },
            { citizen: "かぶとむし", wolf: "くわがた" },
            { citizen: "テレビ", wolf: "ゲーム" },
            { citizen: "ピアノ", wolf: "たいこ" },
            { citizen: "ばなな", wolf: "いちご" },
            { citizen: "おにぎり", wolf: "さんどいっち" }
        ];

        // 低学年向け質問ヒント
        const HINT_QUESTIONS = [
            "「それ は たべもの ですか？」",
            "「どんな いろ を していますか？」",
            "「どれくらいの おおきさ ですか？」",
            "「おうち に ありますか？」",
            "「すきですか？ きらいですか？」",
            "「あさ・ひる・ばん、いつ つかいますか？」",
            "「どこで みかける ことが おおいですか？」",
            "「さわると どんな かんじ が しますか？」",
            "「おとうさん・おかあさん も つかいますか？」"
        ];

        // ゲーム状態
        let playerCount = 3;
        let players = [];
        let currentRevealIndex = 0;
        let citizenWord = "";
        let wolfWord = "";
        let wolfIndex = -1;
        let timerInterval = null;
        let timeLeft = 180; // 3分
        let isTimerRunning = false;

        // Web Audio API（効果音再生）
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
        }

        function playSound(type) {
            try {
                initAudio();
                if (!audioCtx) return;

                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                const now = audioCtx.currentTime;

                if (type === 'tap') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.exponentialRampToValueAtTime(880, now + 0.1);
                    gain.gain.setValueAtTime(0.3, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.1);
                    osc.start(now);
                    osc.stop(now + 0.1);
                } else if (type === 'reveal') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(523.25, now);
                    osc.frequency.setValueAtTime(659.25, now + 0.1);
                    osc.frequency.setValueAtTime(783.99, now + 0.2);
                    gain.gain.setValueAtTime(0.4, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                } else if (type === 'fanfare') {
                    [523.25, 659.25, 783.99, 1046.50].forEach((freq, idx) => {
                        const o = audioCtx.createOscillator();
                        const g = audioCtx.createGain();
                        o.connect(g);
                        g.connect(audioCtx.destination);
                        o.frequency.setValueAtTime(freq, now + idx * 0.12);
                        g.gain.setValueAtTime(0.3, now + idx * 0.12);
                        g.gain.linearRampToValueAtTime(0.01, now + idx * 0.12 + 0.25);
                        o.start(now + idx * 0.12);
                        o.stop(now + idx * 0.12 + 0.25);
                    });
                } else if (type === 'alarm') {
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(880, now);
                    gain.gain.setValueAtTime(0.3, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.3);
                    osc.start(now);
                    osc.stop(now + 0.3);
                }
            } catch (e) {
                console.log("Audio play error", e);
            }
        }

        // 初期化
        window.onload = function() {
            renderPlayerInputs();
        };

        // 人数変更
        function changePlayerCount(delta) {
            playSound('tap');
            const newCount = playerCount + delta;
            if (newCount >= 3 && newCount <= 8) {
                playerCount = newCount;
                document.getElementById('player-count-display').innerText = playerCount;
                renderPlayerInputs();
            }
        }

        // プレイヤー名前入力欄の描画
        function renderPlayerInputs() {
            const container = document.getElementById('player-inputs');
            container.innerHTML = '';
            const defaultNames = ['たろう', 'はなこ', 'じろう', 'さくら', 'けんた', 'あおい', 'ゆうき', 'りん'];

            for (let i = 0; i < playerCount; i++) {
                const div = document.createElement('div');
                div.className = 'flex items-center gap-1 bg-white p-2 rounded-xl border border-sky-100 shadow-sm';
                div.innerHTML = `
                    <span class="text-xs font-bold text-sky-500 w-5 text-center">${i + 1}</span>
                    <input type="text" id="player-name-${i}" value="${defaultNames[i] || 'プレイヤー' + (i + 1)}" 
                        class="w-full text-sm font-bold text-gray-700 bg-transparent focus:outline-none focus:ring-1 focus:ring-sky-300 rounded px-1" />
                `;
                container.appendChild(div);
            }
        }

        // ゲーム開始
        function startGame() {
            playSound('tap');
            players = [];
            for (let i = 0; i < playerCount; i++) {
                const input = document.getElementById(`player-name-${i}`);
                const name = input ? input.value.trim() : `プレイヤー${i + 1}`;
                players.push({ name: name || `プレイヤー${i + 1}`, role: 'citizen', word: '' });
            }

            // おだいペアをランダム選択
            const randomPair = WORD_PAIRS[Math.floor(Math.random() * WORD_PAIRS.length)];
            if (Math.random() < 0.5) {
                citizenWord = randomPair.citizen;
                wolfWord = randomPair.wolf;
            } else {
                citizenWord = randomPair.wolf;
                wolfWord = randomPair.citizen;
            }

            // ウルフを1名ランダム選出
            wolfIndex = Math.floor(Math.random() * playerCount);
            players.forEach((p, idx) => {
                if (idx === wolfIndex) {
                    p.role = 'wolf';
                    p.word = wolfWord;
                } else {
                    p.role = 'citizen';
                    p.word = citizenWord;
                }
            });

            // 画面切り替え（おだい確認へ）
            currentRevealIndex = 0;
            showScreen('screen-reveal');
            setupRevealStep();
        }

        // おだい確認ステップのセットアップ
        function setupRevealStep() {
            document.getElementById('current-player-name').innerText = `${players[currentRevealIndex].name} さん`;
            document.getElementById('reveal-progress').innerText = currentRevealIndex + 1;
            document.getElementById('reveal-total').innerText = playerCount;

            document.getElementById('card-cover').classList.remove('hidden');
            document.getElementById('card-content').classList.add('hidden');
        }

        // カードを開く
        function revealWord() {
            playSound('reveal');
            document.getElementById('secret-word-display').innerText = players[currentRevealIndex].word;
            document.getElementById('card-cover').classList.add('hidden');
            document.getElementById('card-content').classList.remove('hidden');
        }

        // わかりました（次のひとへ）
        function confirmWord() {
            playSound('tap');
            currentRevealIndex++;
            if (currentRevealIndex < playerCount) {
                setupRevealStep();
            } else {
                startTalkPhase();
            }
        }

        // 話し合いフェーズ
        function startTalkPhase() {
            showScreen('screen-talk');
            resetTimer();
            startTimer();
        }

        function resetTimer() {
            clearInterval(timerInterval);
            timeLeft = 180; // 3分
            isTimerRunning = false;
            updateTimerDisplay();
            document.getElementById('btn-timer-toggle').innerText = '▶️ すたーと';
        }

        function updateTimerDisplay() {
            const mins = Math.floor(timeLeft / 60);
            const secs = timeLeft % 60;
            document.getElementById('timer-display').innerText = 
                `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
        }

        function toggleTimer() {
            playSound('tap');
            if (isTimerRunning) {
                clearInterval(timerInterval);
                isTimerRunning = false;
                document.getElementById('btn-timer-toggle').innerText = '▶️ すたーと';
            } else {
                startTimer();
            }
        }

        function startTimer() {
            isTimerRunning = true;
            document.getElementById('btn-timer-toggle').innerText = '⏸️ とめる';
            timerInterval = setInterval(() => {
                if (timeLeft > 0) {
                    timeLeft--;
                    updateTimerDisplay();
                } else {
                    clearInterval(timerInterval);
                    isTimerRunning = false;
                    playSound('alarm');
                    document.getElementById('timer-display').innerText = "00:00";
                }
            }, 1000);
        }

        function getRandomHint() {
            playSound('tap');
            const hint = HINT_QUESTIONS[Math.floor(Math.random() * HINT_QUESTIONS.length)];
            document.getElementById('hint-box').innerText = hint;
        }

        // 投票画面へ
        function goToVote() {
            playSound('tap');
            clearInterval(timerInterval);
            showScreen('screen-vote');

            const container = document.getElementById('vote-buttons');
            container.innerHTML = '';

            players.forEach((p, idx) => {
                const btn = document.createElement('button');
                btn.className = "btn-pop bg-white hover:bg-purple-100 text-purple-900 border-2 border-purple-300 font-black text-lg py-4 px-3 rounded-2xl shadow flex items-center justify-center gap-2";
                btn.onclick = () => selectVotedPlayer(idx);
                btn.innerText = `👤 ${p.name}`;
                container.appendChild(btn);
            });
        }

        // 被疑者の選択・結果発表
        function selectVotedPlayer(votedIdx) {
            playSound('tap');
            showScreen('screen-result');

            const isWolfCaught = (votedIdx === wolfIndex);

            const banner = document.getElementById('result-banner');
            const emoji = document.getElementById('result-emoji');
            const title = document.getElementById('result-title');
            const subtitle = document.getElementById('result-subtitle');

            if (isWolfCaught) {
                playSound('fanfare');
                triggerConfetti();
                banner.className = "bg-gradient-to-r from-emerald-400 to-teal-500 rounded-3xl p-5 card-shadow space-y-2";
                emoji.innerText = "🎊";
                title.innerText = "みんな の かち！";
                subtitle.innerText = `うそつき（わーどうるふ）の 【${players[wolfIndex].name}】 さん を みつけだしたよ！`;
            } else {
                playSound('fanfare');
                banner.className = "bg-gradient-to-r from-rose-400 to-pink-500 rounded-3xl p-5 card-shadow space-y-2";
                emoji.innerText = "🐺";
                title.innerText = "わーどうるふ の かち！";
                subtitle.innerText = `【${players[votedIdx].name}】 さん は うそつき ではありませんでした！`;
            }

            document.getElementById('result-citizen-word').innerText = citizenWord;
            document.getElementById('result-wolf-word').innerText = wolfWord;
            document.getElementById('result-wolf-name').innerText = `${players[wolfIndex].name} さん`;
        }

        // 紙吹雪エフェクト
        function triggerConfetti() {
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 100,
                    spread: 70,
                    origin: { y: 0.6 }
                });
            }
        }

        // 再スタート
        function restartGame() {
            playSound('tap');
            showScreen('screen-setup');
        }

        // 画面切り替えヘルパー
        function showScreen(screenId) {
            const screens = ['screen-setup', 'screen-reveal', 'screen-talk', 'screen-vote', 'screen-result'];
            screens.forEach(id => {
                const el = document.getElementById(id);
                if (id === screenId) {
                    el.classList.remove('hidden');
                } else {
                    el.classList.add('hidden');
                }
            });
        }
    </script>
